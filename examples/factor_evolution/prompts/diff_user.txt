    # 短期反转 + 低波动率复合因子（正IC方向）
    # 1. 短期反转（取负 → 正IC）
    reversal = -data['close'].pct_change(5)
    # 2. 低波动率（取负 → 正IC）
    volatility = -data['close'].pct_change(1).rolling(20).std()
    # 3. 标准化后等权组合
    def ts_z(s, w=60):
        return (s - s.rolling(w, min_periods=30).mean()) / (s.rolling(w, min_periods=30).std() + 1e-8)
    factor = 0.5 * ts_z(reversal) + 0.5 * ts_z(volatility)
    return factor.clip(-3, 3)
